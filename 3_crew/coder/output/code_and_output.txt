# Calculate the first 10,000 terms of the series 1 - 1/3 + 1/5 - 1/7 + ...
total = 0.0
for i in range(10000):
    term = 1 / (2 * i + 1)  # Generates the odd numbers 1, 3, 5, 7, ...
    if i % 2 == 0:  # Even index: add
        total += term
    else:  # Odd index: subtract
        total -= term

result = total * 4
print(result)
